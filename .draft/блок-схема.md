# Блок-схема

Надо предусмотреть логику мульти-опроса.

```mermaid
flowchart TD
    %% ================= БЛОК 1: ИДЕЯ -> ПРОМПТ =================
    A(["Идея"]) --> B["Промпт-инженер"] --> C{"Понял?"}

    C -->|Нет| D1

    subgraph D["Опрос:"]
        direction TB
        D1["☐ Уточнение 1"]
        D2["☐ Уточнение 2"]
        D3["☐ Другое: ___________"]
        D1 ~~~ D2
        D2 ~~~ D3
    end

    D1 --> E(["Промпт"])
    D2 --> E
    D3 -.->|требуется пояснение| C
    C -->|Да| E

    %% не даёт "Промпт" всплыть на уровень D3
    D3 ~~~ E

    %% ================= БЛОК 2: ПРОМПТ -> СПЕЦИФИКАЦИЯ =================
    E --> F["Спек-инженер<br />(CEO/менеджер/промпт-инженер?)"] --> G{"Вводных<br />достаточно?"}

    G -->|Нет| S1

    subgraph S["Опрос:"]
        direction TB
        S1["☐ Уточнение 1"]
        S2["☐ Уточнение 2"]
        S3["☐ Другое: ___________"]
        S1 ~~~ S2
        S2 ~~~ S3
    end

    S1 --> R(["Артефакты"])
    S2 --> R
    S3 -.->|нужны доп. вводные| G
    G -->|Да| R

    %% не даёт "Артефакты" всплыть на уровень S3
    S3 ~~~ R

    %% ================= БЛОК 3: СПЕЦИФИКАЦИЯ -> ИМПЛЕМЕНТАЦИЯ =================
    R -->|сохранить контекст| M[(Память)]
    R -->|сформировать документ| H(["Спецификация"])
    %% H --> I(["План"]) --> J(["Задачи"]) --> K["Имплементация"]
    H --> I(["План"])
    I --> J(["Задачи"])
    J --> K["Имплементация"]

    %% память питает следующие итерации
    M -.->|переиспользуется| B
    M -.->|переиспользуется| F

    %% не даёт "Память" всплыть на уровень "Артефакты"
    R ~~~ M

    %% --- оформление ---
    classDef item fill:#FFFFFF,stroke:#6C757D,stroke-width:1px,color:#212529
    classDef other fill:#FFF9DB,stroke:#6C757D,stroke-width:1px,stroke-dasharray:3 2,color:#212529
    classDef work fill:#D3F9D8,stroke:#2B8A3E,stroke-width:2px,color:#212529
    classDef hub fill:#E5DBFF,stroke:#5F3DC4,stroke-width:2px,color:#212529
    classDef store fill:#FFE8CC,stroke:#D9480F,stroke-width:2px,color:#212529
    class D1,D2,S1,S2 item
    class D3,S3 other
    class M store
    class K work
    style D fill:#F8F9FA,stroke:#495057,stroke-width:1.5px,color:#212529
    style S fill:#F8F9FA,stroke:#495057,stroke-width:1.5px,color:#212529
```
